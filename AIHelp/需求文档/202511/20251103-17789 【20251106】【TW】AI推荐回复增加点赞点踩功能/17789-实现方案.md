```
CREATE TABLE `ai_ticket_copilot_message_record`  (

  `id` varchar(50) NOT NULL DEFAULT '' COMMENT '主键ID',

  `accountId` int(11) NOT NULL DEFAULT 0 COMMENT '租户ID',

  `gameId` int(11) NOT NULL DEFAULT 0 COMMENT '应用ID',

  `ticketId` varchar(48) NOT NULL DEFAULT '' COMMENT '客诉ID',

  `ticketShortId` varchar(20) NOT NULL COMMENT '客诉短ID',

  `ticketBirthTime` bigint(20) NULL DEFAULT 0 COMMENT '客诉创建时间（毫秒时间戳）',

  `messageId` varchar(50) NOT NULL DEFAULT '' COMMENT 'mongodb的AI推荐回复消息ID',

  `userMessage` text NULL DEFAULT NULL COMMENT '玩家消息',

  `userMessageTime` bigint(20) NULL DEFAULT NULL COMMENT '玩家消息时间',

  `question` text NULL DEFAULT NULL COMMENT 'AI推荐回复问题',

  `answer` text NULL DEFAULT NULL COMMENT 'AI推荐回复答案',

  `copilotMessage` text NULL DEFAULT NULL COMMENT 'AI推荐回复内容（json，结构参考AI助手回复）',

  `knowledgeRefCount` int(11) NOT NULL DEFAULT 0 COMMENT '回复时参考的知识片段数',

  `platform` varchar(32) NOT NULL DEFAULT '' COMMENT '平台',

  `languageId` int(11) NULL DEFAULT NULL COMMENT '语言',

  `feedbackType` int(11) NULL DEFAULT NULL COMMENT '赞踩类型：1=赞，2=踩，NULL无反馈',

  `feedbackTime` bigint(20) NULL DEFAULT NULL COMMENT '反馈时间（时间戳，毫秒）',

  `feedbackuserId` int(11) NULL DEFAULT NULL COMMENT '反馈客服ID',

  `createTime` bigint(20) NOT NULL DEFAULT 0 COMMENT '记录创建时间（毫秒时间戳）',

  `isReference` tinyint(1) NULL DEFAULT NULL COMMENT '是否引用：0-否；1-是',

  `referenceTime` bigint(20) NULL DEFAULT NULL COMMENT '引用时间',

  `referenceUserId` int(11) NULL DEFAULT NULL COMMENT '引用操作人',

  PRIMARY KEY (`id`) USING BTREE,

  INDEX `idx_accountId_ticketId`(`accountId`, `ticketId`) USING BTREE,

  INDEX `idx_accountId_ticketShortId`(`accountId`, `ticketShortId`) USING BTREE,

  INDEX `idx_accountId_messageId`(`accountId`, `messageId`) USING BTREE,

  INDEX `idx_accountId_gameId_createTime_languageId_platform_feedbackType`(`accountId`, `gameId`, `createTime`, `languageId`, `feedbackType`, `platform`) USING BTREE

) ENGINE = InnoDB COMMENT = 'AI推荐回复记录' ROW_FORMAT = Dynamic;
```
