CREATE TABLE `aigc_record`  (
  `id` bigint(20) NOT NULL AUTO_INCREMENT,
  `gameId` int(11) NOT NULL,
  `accountId` int(11) NOT NULL,
  `ticketId` varchar(50) NULL DEFAULT NULL COMMENT '客诉id',
  `chatTimeSpan` bigint(20) NOT NULL,
  `promptTokens` int(11) NULL DEFAULT NULL COMMENT '请求token数',
  `completionTokens` int(11) NULL DEFAULT NULL COMMENT '响应token数',
  `totalTokens` int(11) NULL DEFAULT NULL COMMENT '总共消耗token数',
  `request` text NULL COMMENT '请求内容',
  `response` text NULL COMMENT '返回内容',
  `module` int(11) NULL DEFAULT 0 COMMENT '0.RPA问答机器人；1.人工客诉；2.FAQ',
  `moduleType` int(11) NULL DEFAULT 0 COMMENT '1.扩展；2.重写；3.更正式；4.更友好;5.FAQ相似问题',
  `templateId` int(11) NULL DEFAULT 0 COMMENT '模板id',
  `userId` int(11) NULL DEFAULT 0 COMMENT '客服id',
  `aigcType` int(11) NULL DEFAULT 1 COMMENT '使用的服务类型：1.gpt-3.5-turbo;2.ada-v2',
  PRIMARY KEY (`id`) USING BTREE,
  INDEX `idx_chatTimeSpan_accountId_gameId_ticketId`(`chatTimeSpan`, `accountId`, `gameId`, `ticketId`) USING BTREE,
  INDEX `idx_chatTimeSpan_module_moduleType`(`chatTimeSpan`, `module`, `moduleType`) USING BTREE
) ENGINE = InnoDB ROW_FORMAT = Dynamic;
